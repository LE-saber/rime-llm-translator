cmake_minimum_required(VERSION 3.22)
project(rime_mt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(GIT_SUBMODULE OFF CACHE BOOL "" FORCE)
set(BUILD_ARCH x86-64 CACHE STRING "" FORCE)
set(COMPILE_CUDA OFF CACHE BOOL "" FORCE)
set(COMPILE_CPU ON CACHE BOOL "" FORCE)
set(COMPILE_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_PYTHON OFF CACHE BOOL "" FORCE)
set(COMPILE_SERVER OFF CACHE BOOL "" FORCE)
set(USE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(USE_WASM_COMPATIBLE_SOURCE ON CACHE BOOL "" FORCE)
set(USE_INTGEMM ON CACHE BOOL "" FORCE)
set(USE_RUY OFF CACHE BOOL "" FORCE)
set(USE_RUY_SGEMM OFF CACHE BOOL "" FORCE)
set(USE_MKL OFF CACHE BOOL "" FORCE)
set(USE_STATIC_LIBS OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
add_subdirectory("${MOZILLA_SOURCE}/inference" bergamot EXCLUDE_FROM_ALL)
find_package(nlohmann_json REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENCC REQUIRED IMPORTED_TARGET opencc)
add_executable(rime-mt main.cpp)
target_compile_features(rime-mt PRIVATE cxx_std_17)
target_link_libraries(rime-mt PRIVATE bergamot-translator-source nlohmann_json::nlohmann_json PkgConfig::OPENCC)
